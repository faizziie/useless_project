import tkinter as tk from tkinter import messagebox import threading import time import datetime import numpy as np import sounddevice as sd class ScreamAlarm: def __init__(self, root): self.root = root self.root.title("Scream-to-Dismiss Alarm") self.root.geometry("450x500") self.root.resizable(False, False) self.alarm_active = False self.alarm_thread = None self.monitoring = False # Settings self.scream_duration = 3.0 # seconds self.volume_threshold = 0.20 # microphone loudness threshold # ----------------------------- # UI # ----------------------------- title = tk.Label( root, text="SCREAM ALARM", font=("Arial", 24, "bold") ) title.pack(pady=20) # Alarm time tk.Label( root, text="Set alarm time (24-hour format)", font=("Arial", 12) ).pack() self.time_entry = tk.Entry( root, font=("Arial", 20), justify="center" ) self.time_entry.insert(0, "07:00") self.time_entry.pack(pady=10) # Scream duration tk.Label( root, text="Scream duration (seconds)", font=("Arial", 12) ).pack(pady=(15, 0)) self.duration_entry = tk.Entry( root, font=("Arial", 16), justify="center" ) self.duration_entry.insert(0, "3") self.duration_entry.pack(pady=10) # Volume threshold tk.Label( root, text="Scream sensitivity", font=("Arial", 12) ).pack(pady=(15, 0)) self.threshold_scale = tk.Scale( root, from_=0.05, to=0.8, resolution=0.01, orient="horizontal", length=250 ) self.threshold_scale.set(self.volume_threshold) self.threshold_scale.pack() # Buttons self.set_button = tk.Button( root, text="SET ALARM", font=("Arial", 14, "bold"), command=self.set_alarm, width=18 ) self.set_button.pack(pady=20) self.stop_button = tk.Button( root, text="CANCEL ALARM", font=("Arial", 12), command=self.cancel_alarm, width=18, state=tk.DISABLED ) self.stop_button.pack() # Status self.status_label = tk.Label( root, text="No alarm set", font=("Arial", 12) ) self.status_label.pack(pady=20) # Volume meter self.volume_label = tk.Label( root, text="Microphone: --", font=("Arial", 12) ) self.volume_label.pack() self.volume_bar = tk.Scale( root, from_=0, to=1, resolution=0.01, orient="horizontal", length=300, showvalue=False ) self.volume_bar.pack(pady=10) # ----------------------------- # Set alarm # ----------------------------- def set_alarm(self): alarm_time = self.time_entry.get().strip() try: datetime.datetime.strptime(alarm_time, "%H:%M") except ValueError: messagebox.showerror( "Invalid time", "Please enter the time in HH:MM format.\nExample: 07:30" ) return try: duration = float(self.duration_entry.get()) if duration <= 0: raise ValueError except ValueError: messagebox.showerror( "Invalid duration", "Scream duration must be a positive number." ) return self.scream_duration = duration self.volume_threshold = self.threshold_scale.get() self.alarm_active = True self.set_button.config(state=tk.DISABLED) self.stop_button.config(state=tk.NORMAL) self.status_label.config( text=f"Alarm set for {alarm_time}" ) self.alarm_thread = threading.Thread( target=self.wait_for_alarm, args=(alarm_time,), daemon=True ) self.alarm_thread.start() # ----------------------------- # Wait for alarm # ----------------------------- def wait_for_alarm(self, alarm_time): while self.alarm_active: now = datetime.datetime.now() current_time = now.strftime("%H:%M") if current_time == alarm_time: self.start_alarm() break time.sleep(1) # ----------------------------- # Start alarm # ----------------------------- def start_alarm(self): self.root.after( 0, lambda: self.status_label.config( text="🚨 ALARM! SCREAM TO DISMISS!" ) ) # Start microphone monitoring self.monitoring = True microphone_thread = threading.Thread( target=self.monitor_microphone, daemon=True ) microphone_thread.start() # Keep alarm sound running sound_thread = threading.Thread( target=self.alarm_sound, daemon=True ) sound_thread.start() # ----------------------------- # Alarm sound # ----------------------------- def alarm_sound(self): while self.alarm_active and self.monitoring: try: frequency = 1000 duration = 0.5 sample_rate = 44100 samples = np.arange( int(sample_rate * duration) ) wave = 0.3 * np.sin( 2 * np.pi * frequency * samples / sample_rate ) sd.play(wave, sample_rate) sd.wait() except Exception as e: print("Sound error:", e) break time.sleep(0.2) # ----------------------------- # Microphone monitoring # ----------------------------- def monitor_microphone(self): loud_start = None try: with sd.InputStream( channels=1, samplerate=44100, blocksize=2048 ) as stream: while self.alarm_active and self.monitoring: audio_data, overflowed = stream.read(2048) # Convert microphone signal to numpy array audio = audio_data[:, 0] # Calculate RMS volume volume = np.sqrt( np.mean(np.square(audio)) ) # Limit value for UI display_volume = min(volume * 5, 1) self.root.after( 0, lambda v=display_volume: self.volume_bar.set(v) ) self.root.after( 0, lambda v=volume: self.volume_label.config( text=f"Microphone: {v:.3f}" ) ) # Check whether user is screaming if volume >= self.volume_threshold: if loud_start is None: loud_start = time.time() elapsed = time.time() - loud_start self.root.after( 0, lambda e=elapsed: self.status_label.config( text=f"Keep screaming! " f"{e:.1f}/{self.scream_duration:.1f}s" ) ) # Scream duration reached if elapsed >= self.scream_duration: self.dismiss_alarm() break else: # Volume dropped below threshold loud_start = None self.root.after( 0, lambda: self.status_label.config( text="🚨 SCREAM TO DISMISS!" ) ) except Exception as e: print("Microphone error:", e) self.root.after( 0, lambda: messagebox.showerror( "Microphone Error", f"Could not access the microphone.\n\n{e}" ) ) # ----------------------------- # Dismiss alarm # ----------------------------- def dismiss_alarm(self): self.monitoring = False self.alarm_active = False sd.stop() self.root.after( 0, lambda: self.status_label.config( text="✅ Alarm dismissed!" ) ) self.root.after( 0, lambda: self.set_button.config( state=tk.NORMAL ) ) self.root.after( 0, lambda: self.stop_button.config( state=tk.DISABLED ) ) # ----------------------------- # Cancel alarm # ----------------------------- def cancel_alarm(self): self.alarm_active = False self.monitoring = False sd.stop() self.status_label.config( text="Alarm cancelled" ) self.set_button.config( state=tk.NORMAL ) self.stop_button.config( state=tk.DISABLED ) # ----------------------------- # Run application # ----------------------------- if __name__ == "__main__": root = tk.Tk() app = ScreamAlarm(root) root.mainloop()
